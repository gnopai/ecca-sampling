\begin{frame}
  \frametitle{Sampling Design}
  \textbf{Approach:}
  \begin{itemize}
    \pause
    \item Determine areas (strata) to use to group all the highways and ramps
    \pause
    \item Assumption: trash in a given area is relatively similar anywhere in the area, but possibly different from other areas
    \pause
    \item Pick some number of segments from each area to use in the sample
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Groupings}
  \textbf{Approach:}
  \begin{itemize}
    \pause
    \item 5 highway areas and 5 samples: take 1 sample from each
    \pause
    \item 3 ramp pairs and 3 samples: take 1 sample from each pair
    \pause
    \item Make sure at least 1 on-ramp and 1 off-ramp included
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Sampling Execution}
  \begin{block}{Idea}
    Do sampling entirely in software using geospatial data from WSDOT
  \end{block}
  \bigskip
  \bigskip
  \pause
  We use data from the \textbf{WSDOT Geospatial Open Data Portal}:
  \begin{itemize}
    \item Highway data: "WSDOT - State Route Linear Referencing System (LRS) Current"
    \item Ramp data: "WSDOT - State Route Ramp (1:24K) Current"
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Highway Sampling}
  \textbf{Approach:}
  \begin{enumerate}
    \pause
    \item Load geospatial data for both sides of Highway 14
    \pause
    \item Filter these into the 5 areas using the milepost boundaries for what ECCA cleans
    \pause
    \item Divide each area into 600 foot segments
    \pause
    \item Randomly select one of these segments in each area
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Ramp Sampling}
  \begin{block}{Idea}
    Pick one ramp from each pair, but make sure at least one on-ramp and one off-ramp are selected
  \end{block}
\end{frame}
